"""Generation two of the fault localizer.

Evidence, combined per production line:
  * production stack frames (exact lines, decaying weight);
  * identifiers reached from the failing test: its body, test helpers it calls (one hop),
    and test-side frames; each weighted by rarity across production files (IDF), so names
    that exist only in test or framework code contribute nothing;
  * per line: weighted hits of those identifiers on the line itself;
  * per method: coverage of the test's identifiers by the enclosing method, and declaration
    matches (name, parameters, test-method tokens, frame method names);
  * per file: share of the test's identifiers the file uses, plus class-name matches.
Selection is greedy with a 20-line suppression radius per file, six entries at most.
Work is bounded by a deterministic budget on files and bytes read, not by the clock.
"""
import bisect
import math
import os
import re

KEYWORDS = {
    "abstract", "assert", "boolean", "break", "byte", "case", "catch", "char", "class", "const",
    "continue", "default", "do", "double", "else", "enum", "extends", "final", "finally", "float",
    "for", "goto", "if", "implements", "import", "instanceof", "int", "interface", "long", "native",
    "new", "package", "private", "protected", "public", "return", "short", "static", "strictfp",
    "super", "switch", "synchronized", "this", "throw", "throws", "transient", "try", "void",
    "volatile", "while", "true", "false", "null", "var", "yield",
}
NOT_TYPE = {"return", "new", "throw", "else", "case", "yield", "throws", "import", "package",
            "assert", "extends", "implements", "instanceof"}
STOP = {"test", "tests", "the", "and", "with", "for", "not", "should", "can", "was", "when", "from",
        "into", "that", "this", "has", "have", "are", "but", "all"}

FRAME = re.compile(r"at\s+([\w.$]+)\.([\w$<>-]+)\(([^():\s]+):(\d+)\)")
DECL = re.compile(
    r"^\s*(?:@[\w.]+(?:\([^)]*\))?\s+)*"
    r"(?:(?:public|protected|private|static|final|abstract|synchronized|native|default|strictfp)\s+)*"
    r"(?:<[^>]*>\s*)?"
    r"([\w.$]+(?:<[^;=(){}]*>)?(?:\[\])*)\s+([A-Za-z_$][\w$]*)\s*\(([^)]*)")
CTOR = re.compile(r"^\s*(?:(?:public|protected|private)\s+)?([A-Z][\w$]*)\s*\(([^)]*)")
CLASS = re.compile(r"\b(?:class|interface|enum)\s+([A-Za-z_$][\w$]*)")
IDENT = re.compile(r"[A-Za-z_$][\w$]*")
TOKEN = re.compile(r"[A-Z]+(?![a-z])|[A-Z]?[a-z]+|\d+")
EMPTY = frozenset()

MAX_FILES = 4000
MAX_BYTES = 20_000_000
SPAN_CAP = 80


def tokens(s):
    return {t.lower() for t in TOKEN.findall(s) if len(t) >= 3 and t.lower() not in STOP}


def parse(text, base):
    lines = text.splitlines()
    decls = []
    classes = {}
    for i, raw in enumerate(lines, 1):
        s = raw.strip()
        if not s or s.startswith(("*", "/*", "//")):
            continue
        code = raw.split("//", 1)[0]
        m = CLASS.search(code)
        if m:
            classes.setdefault(m.group(1), i)
        m = DECL.match(code)
        if m:
            typ = m.group(1).split("<")[0].split(".")[-1]
            name = m.group(2)
            if typ not in NOT_TYPE and name not in KEYWORDS:
                decls.append((i, name, m.group(3)))
                continue
        m = CTOR.match(code)
        if m and m.group(1) == base:
            decls.append((i, base, m.group(2)))
    ids = []
    alls = set()
    for raw in lines:
        found = IDENT.findall(raw)
        if found:
            fs = frozenset(found)
            alls |= fs
            ids.append(fs)
        else:
            ids.append(EMPTY)
    return {"ids": ids, "decls": decls, "starts": [d[0] for d in decls], "base": base,
            "classes": classes, "all": alls, "n": len(lines)}


def span(p, k):
    starts = p["starts"]
    s = starts[k]
    nxt = starts[k + 1] if k + 1 < len(starts) else p["n"] + 1
    e = min(nxt, s + SPAN_CAP)
    return s, max(e, s + 1)


def ids_in(p, s, e):
    out = set()
    for fs in p["ids"][s - 1:e - 1]:
        out |= fs
    return out


def line_ids(p, line):
    k = bisect.bisect_right(p["starts"], line) - 1
    if k < 0:
        s, e = max(1, line - 10), line + 1
    else:
        s, e = span(p, k)
    e = max(e, line + 1)
    return ids_in(p, s, e)


def put(d, names, w):
    for n in names:
        if d.get(n, 0.0) < w:
            d[n] = w


def localize(case):
    root = case["root"]
    src = case.get("source_directory", "") or ""
    tst = case.get("test_directory", "") or ""
    report = case.get("report", "") or ""
    failing = case.get("failing_tests", []) or []

    budget = {"files": 0, "bytes": 0}

    def index(base):
        out = {}
        if not base:
            return out
        full = os.path.join(root, base)
        if not os.path.isdir(full):
            return out
        for dp, dn, fn in os.walk(full):
            dn.sort()
            for f in sorted(fn):
                if not f.endswith(".java"):
                    continue
                if budget["files"] >= MAX_FILES or budget["bytes"] >= MAX_BYTES:
                    return out
                p = os.path.join(dp, f)
                try:
                    with open(p, encoding="utf-8", errors="replace") as h:
                        text = h.read(1000000)
                except OSError:
                    continue
                budget["files"] += 1
                budget["bytes"] += len(text)
                rel = os.path.relpath(p, root).replace(os.sep, "/")
                out[rel] = parse(text, f[:-5])
        return out

    prod = index(src)
    tests = index(tst)

    def resolve(fqn, fname, idx):
        top = fqn.split("$")[0]
        suffix = top.replace(".", "/") + ".java"
        matches = [k for k in idx if k == suffix or k.endswith("/" + suffix)]
        if matches:
            return min(matches, key=lambda k: (len(k), k))
        by_name = [k for k in idx if k.endswith("/" + fname)]
        if len(by_name) == 1:
            return by_name[0]
        return None

    frames = FRAME.findall(report)
    report_ids = set(IDENT.findall(report))
    frame_methods = {m for _, m, _, _ in frames if re.fullmatch(r"[A-Za-z_$][\w$]*", m)}

    # test-side identifiers with weights
    tdecl = {}
    for rel, p in tests.items():
        for k, (_, name, _) in enumerate(p["decls"]):
            tdecl.setdefault(name, []).append((rel, k))

    A = {}
    T = set()
    test_classes = set()
    for t in failing:
        cls_part, _, meth = t.partition("::")
        simple = cls_part.split(".")[-1].split("$")[0]
        test_classes.add(simple)
        if not meth:
            continue
        T |= tokens(meth)
        for rel, p in tests.items():
            if os.path.basename(rel) != simple + ".java":
                continue
            for k, (_, name, _) in enumerate(p["decls"]):
                if name != meth:
                    continue
                s, e = span(p, k)
                ids = ids_in(p, s, e)
                put(A, ids, 1.0)
                for h in ids:
                    if h != meth and len(tdecl.get(h, ())) == 1:
                        hr, hk = tdecl[h][0]
                        hp = tests[hr]
                        hs, he = span(hp, hk)
                        put(A, ids_in(hp, hs, he), 0.5)
    for fqn, meth, fname, ln in frames:
        rel = resolve(fqn, fname, tests)
        if rel:
            put(A, line_ids(tests[rel], int(ln)), 1.0)

    # rarity weights against production files
    N = max(1, len(prod))
    df = {}
    for p in prod.values():
        for n in p["all"]:
            df[n] = df.get(n, 0) + 1
    W = {}
    for n, a in A.items():
        d = df.get(n, 0)
        if d == 0 or n in KEYWORDS or len(n) < 2:
            continue
        W[n] = a * (math.log((N + 1) / (d + 1)) + 0.2)
    scale = max(W.values()) if W else 1.0
    wsum = sum(W.values()) or 1.0

    cands = {}

    def add(key, score):
        if score > cands.get(key, -1.0):
            cands[key] = score

    # production stack frames
    rank = 0
    seen = set()
    for fqn, meth, fname, ln in frames:
        rel = resolve(fqn, fname, prod)
        if not rel:
            continue
        key = (rel, int(ln))
        if key in seen:
            continue
        seen.add(key)
        add(key, max(0.6, 3.0 - 0.3 * rank))
        rank += 1

    # per-file scoring: lines, declarations, method coverage
    for rel, p in prod.items():
        base = p["base"]
        n = p["n"]
        stripped_hit = any(base == re.sub(r"Tests?$", "", tc) for tc in test_classes if tc)
        fscore = sum(w for x, w in W.items() if x in p["all"]) / wsum
        fb = 2.0 * fscore + (0.5 if base in W else 0.0) + (0.5 if stripped_hit else 0.0)

        cover = [0.0] * (n + 2)
        dscore = {}
        for k, (line, name, params) in enumerate(p["decls"]):
            s, e = span(p, k)
            names = ids_in(p, s, e)
            cov = sum(W[x] for x in names if x in W)
            cov_n = 0.3 * min(cov / scale, 8.0)
            for l in range(s, min(e, n + 1)):
                cover[l] = cov_n
            ds = 0.0
            if name in W:
                ds += W[name] / scale
            ds += 0.8 * (len(tokens(name) & T) + 0.5 * len(tokens(params) & T))
            if name in frame_methods:
                ds += 1.0
            if name in report_ids:
                ds += 0.2
            if ds > 0:
                dscore[line] = max(dscore.get(line, 0.0), ds)

        for l in range(1, n + 1):
            fs = p["ids"][l - 1]
            h = 0.0
            if fs:
                h = sum(W[x] for x in fs if x in W) / scale
            d = dscore.get(l, 0.0)
            c = cover[l]
            if h > 0 or d > 0:
                add((rel, l), h + d + c + fb)
            elif c > 0 and l in dscore_starts(p):
                add((rel, l), c + fb)

    # last-resort fallback: file named after the test class
    if not cands:
        for tc in test_classes:
            stripped = re.sub(r"Tests?$", "", tc)
            for rel in prod:
                if os.path.basename(rel) == stripped + ".java":
                    add((rel, 1), 0.1)

    ordered = sorted(cands.items(), key=lambda kv: (-kv[1], kv[0][0], kv[0][1]))
    chosen = []
    for (p, l), _ in ordered:
        if any(p == cp and abs(l - cl) <= 20 for cp, cl in chosen):
            continue
        chosen.append((p, l))
        if len(chosen) == 6:
            break
    return [[p, l] for p, l in chosen]


def dscore_starts(p):
    return set(p["starts"])
