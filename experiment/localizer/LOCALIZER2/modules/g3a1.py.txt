"""Generation two of the fault localizer.

Signals:
  * production stack frames (innermost first, decaying weight), each also anchored
    to the enclosing production method declaration and to project callees invoked
    from that method;
  * declarations in production sources whose names are called from the failing test
    body or from test frames (weighted by rarity), or appear in the traces;
  * parameter and name tokens of the failing test method matched against declarations;
  * class names referenced by the test, as a file-level fallback.
Selection is greedy: each pick is penalised by the number of earlier picks in the same
file, and a 20-line suppression radius applies per file, so the six slots spread over
distinct places and files.
Work is bounded by a deterministic budget on files and bytes read, not by the clock.
"""
import bisect
import os
import re

KEYWORDS = {
    "abstract", "assert", "boolean", "break", "byte", "case", "catch", "char", "class", "const",
    "continue", "default", "do", "double", "else", "enum", "extends", "final", "finally", "float",
    "for", "goto", "if", "implements", "import", "instanceof", "int", "interface", "long", "native",
    "new", "package", "private", "protected", "public", "return", "short", "static", "strictfp",
    "super", "switch", "synchronized", "this", "throw", "throws", "transient", "try", "void",
    "volatile", "while", "true", "false", "null", "var", "yield",
}
NOT_TYPE = {"return", "new", "throw", "else", "case", "yield", "throws", "import", "package",
            "assert", "extends", "implements", "instanceof"}
STOP = {"test", "tests", "the", "and", "with", "for", "not", "should", "can", "was", "when", "from",
        "into", "that", "this", "has", "have", "are", "but", "all"}

FRAME = re.compile(r"at\s+([\w.$]+)\.([\w$<>-]+)\(([^():\s]+):(\d+)\)")
DECL = re.compile(
    r"^\s*(?:@[\w.]+(?:\([^)]*\))?\s+)*"
    r"(?:(?:public|protected|private|static|final|abstract|synchronized|native|default|strictfp)\s+)*"
    r"(?:<[^>]*>\s*)?"
    r"([\w.$]+(?:<[^;=(){}]*>)?(?:\[\])*)\s+([A-Za-z_$][\w$]*)\s*\(([^)]*)")
CTOR = re.compile(r"^\s*(?:(?:public|protected|private)\s+)?([A-Z][\w$]*)\s*\(([^)]*)")
CLASS = re.compile(r"\b(?:class|interface|enum)\s+([A-Za-z_$][\w$]*)")
IDENT = re.compile(r"[A-Za-z_$][\w$]*")
TOKEN = re.compile(r"[A-Z]+(?![a-z])|[A-Z]?[a-z]+|\d+")

MAX_FILES = 6000
MAX_BYTES = 40_000_000
RADIUS = 20
FILE_PENALTY = 0.25


def tokens(s):
    return {t.lower() for t in TOKEN.findall(s) if len(t) >= 3 and t.lower() not in STOP}


def parse(text, base):
    lines = text.splitlines()
    decls = []
    classes = {}
    for i, raw in enumerate(lines, 1):
        s = raw.strip()
        if not s or s.startswith(("*", "/*", "//")):
            continue
        code = raw.split("//", 1)[0]
        m = CLASS.search(code)
        if m:
            classes.setdefault(m.group(1), i)
        m = DECL.match(code)
        if m:
            typ = m.group(1).split("<")[0].split(".")[-1]
            name = m.group(2)
            if typ not in NOT_TYPE and name not in KEYWORDS:
                decls.append((i, name, m.group(3)))
                continue
        m = CTOR.match(code)
        if m and m.group(1) == base:
            decls.append((i, base, m.group(2)))
    return {"lines": lines, "decls": decls, "base": base, "classes": classes}


def localize(case):
    root = case["root"]
    src = case.get("source_directory", "") or ""
    tst = case.get("test_directory", "") or ""
    report = case.get("report", "") or ""
    failing = case.get("failing_tests", []) or []

    budget = {"files": 0, "bytes": 0}

    def index(base):
        out = {}
        if not base:
            return out
        full = os.path.join(root, base)
        if not os.path.isdir(full):
            return out
        for dp, dn, fn in os.walk(full):
            dn.sort()
            for f in sorted(fn):
                if not f.endswith(".java"):
                    continue
                if budget["files"] >= MAX_FILES or budget["bytes"] >= MAX_BYTES:
                    return out
                p = os.path.join(dp, f)
                try:
                    with open(p, encoding="utf-8", errors="replace") as h:
                        text = h.read(1000000)
                except OSError:
                    continue
                budget["files"] += 1
                budget["bytes"] += len(text)
                rel = os.path.relpath(p, root).replace(os.sep, "/")
                out[rel] = parse(text, f[:-5])
        return out

    prod = index(src)
    tests = index(tst)

    def resolve(fqn, fname, idx):
        top = fqn.split("$")[0]
        suffix = top.replace(".", "/") + ".java"
        matches = [k for k in idx if k == suffix or k.endswith("/" + suffix)]
        if matches:
            return min(matches, key=lambda k: (len(k), k))
        by_name = [k for k in idx if k.endswith("/" + fname)]
        if len(by_name) == 1:
            return by_name[0]
        return None

    def span_text(parsed, line):
        lines = parsed["lines"]
        starts = [d[0] for d in parsed["decls"]]
        idx = bisect.bisect_right(starts, line) - 1
        start = starts[idx] if idx >= 0 else max(1, line - 10)
        end = starts[idx + 1] if idx + 1 < len(starts) else len(lines) + 1
        end = min(end, start + 80)
        end = max(end, line + 1)
        return start, "\n".join(lines[start - 1:end - 1])

    def span_ids(parsed, line):
        return set(IDENT.findall(span_text(parsed, line)[1]))

    def enclosing_decl(parsed, line):
        starts = [d[0] for d in parsed["decls"]]
        idx = bisect.bisect_right(starts, line) - 1
        if idx < 0:
            return None
        return starts[idx]

    frames = FRAME.findall(report)
    report_ids = set(IDENT.findall(report))
    frame_methods = {m for _, m, _, _ in frames if re.fullmatch(r"[A-Za-z_$][\w$]*", m)}

    A = set()          # identifiers used by the failing tests
    T = set()          # tokens of failing test method names
    test_classes = set()
    for t in failing:
        cls_part, _, meth = t.partition("::")
        simple = cls_part.split(".")[-1].split("$")[0]
        test_classes.add(simple)
        if not meth:
            continue
        T |= tokens(meth)
        for k, parsed in tests.items():
            if os.path.basename(k) != simple + ".java":
                continue
            for line, name, _ in parsed["decls"]:
                if name == meth:
                    A |= span_ids(parsed, line)
    for fqn, meth, fname, ln in frames:
        rel = resolve(fqn, fname, tests)
        if rel:
            A |= span_ids(tests[rel], int(ln))

    prod_counts = {}
    for parsed in prod.values():
        for _, name, _ in parsed["decls"]:
            prod_counts[name] = prod_counts.get(name, 0) + 1

    def count(name):
        return max(1, prod_counts.get(name, 1))

    cands = {}

    def add(key, score):
        if score > cands.get(key, -1.0):
            cands[key] = score

    # production stack frames: the frame line, its enclosing method, and project callees
    rank = 0
    seen = set()
    callee_budget = 3
    for fqn, meth, fname, ln in frames:
        rel = resolve(fqn, fname, prod)
        if not rel:
            continue
        parsed = prod[rel]
        line = int(ln)
        w = max(0.3, 2.6 - 0.2 * rank)
        key = (rel, line)
        if key not in seen:
            seen.add(key)
            add(key, w)
            rank += 1
        enc = enclosing_decl(parsed, line)
        if enc is not None:
            add((rel, enc), w * 0.7)
        if callee_budget > 0 and enc is not None:
            callee_budget -= 1
            body = span_text(parsed, line)[1]
            for name in set(IDENT.findall(body)):
                if name in KEYWORDS or name == meth:
                    continue
                if name in prod_counts and name != parsed["base"]:
                    # project callee: anchored at its declaration lines below via name match
                    A.add(name)

    # declarations and class references
    for rel, parsed in prod.items():
        base = parsed["base"]
        for line, name, params in parsed["decls"]:
            c = count(name) ** 0.5
            s = 0.0
            if name in A:
                s += 2.0 / c
            if name in frame_methods:
                s += 1.2 / c
            if name in report_ids:
                s += 0.6 / c
            ov = len(tokens(name) & T) + 0.5 * len(tokens(params) & T)
            s += 0.4 * ov
            if base in A:
                s += 0.5
            if base in report_ids:
                s += 0.3
            if s >= 0.4:
                add((rel, line), s)
        if base in A and base in parsed["classes"]:
            add((rel, parsed["classes"][base]), 0.8)
        for tc in test_classes:
            stripped = re.sub(r"Tests?$", "", tc)
            if stripped and base == stripped:
                add((rel, parsed["classes"].get(base, 1)), 0.2)

    # last-resort fallback: file named after the test class
    if not cands:
        for tc in test_classes:
            stripped = re.sub(r"Tests?$", "", tc)
            for rel in prod:
                if os.path.basename(rel) == stripped + ".java":
                    add((rel, 1), 0.1)

    pool = sorted(cands.items(), key=lambda kv: (-kv[1], kv[0][0], kv[0][1]))
    pool = [((p, l), s) for (p, l), s in pool]
    chosen = []
    file_count = {}
    while pool and len(chosen) < 6:
        best_i = None
        best_val = None
        for i, ((p, l), s) in enumerate(pool):
            if any(p == cp and abs(l - cl) <= RADIUS for cp, cl in chosen):
                continue
            val = s - FILE_PENALTY * file_count.get(p, 0)
            if best_val is None or val > best_val:
                best_val = val
                best_i = i
        if best_i is None:
            break
        (p, l), _ = pool.pop(best_i)
        chosen.append((p, l))
        file_count[p] = file_count.get(p, 0) + 1
    return [[p, l] for p, l in chosen]
