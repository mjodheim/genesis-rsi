"""Generation two of the fault localizer.

Signals:
  * production stack frames (innermost first, decaying weight);
  * line-level overlap between production lines and the failing test: identifiers and string
    literal words from the test body, words of the test method name, frame method names and
    report words. Each shared token is weighted by its rarity in the production corpus;
  * declaration-level signals (names called from the test, frame methods, traces, parameter tokens);
  * a share of the score of the enclosing declaration, so that a method exercised by the test
    lends its score to all of its lines;
  * class-name and file-name fallbacks.
Selection is greedy with a 20-line suppression radius per file, so the six slots cover distinct places.
Work is bounded by a deterministic budget on files and bytes read, not by the clock.
"""
import bisect
import math
import os
import re
from collections import Counter

KEYWORDS = {
    "abstract", "assert", "boolean", "break", "byte", "case", "catch", "char", "class", "const",
    "continue", "default", "do", "double", "else", "enum", "extends", "final", "finally", "float",
    "for", "goto", "if", "implements", "import", "instanceof", "int", "interface", "long", "native",
    "new", "package", "private", "protected", "public", "return", "short", "static", "strictfp",
    "super", "switch", "synchronized", "this", "throw", "throws", "transient", "try", "void",
    "volatile", "while", "true", "false", "null", "var", "yield",
}
NOT_TYPE = {"return", "new", "throw", "else", "case", "yield", "throws", "import", "package",
            "assert", "extends", "implements", "instanceof"}
STOP = {"test", "tests", "the", "and", "with", "for", "not", "should", "can", "was", "when", "from",
        "into", "that", "this", "has", "have", "are", "but", "all"}

FRAME = re.compile(r"at\s+([\w.$]+)\.([\w$<>-]+)\(([^():\s]+):(\d+)\)")
DECL = re.compile(
    r"^\s*(?:@[\w.]+(?:\([^)]*\))?\s+)*"
    r"(?:(?:public|protected|private|static|final|abstract|synchronized|native|default|strictfp)\s+)*"
    r"(?:<[^>]*>\s*)?"
    r"([\w.$]+(?:<[^;=(){}]*>)?(?:\[\])*)\s+([A-Za-z_$][\w$]*)\s*\(([^)]*)")
CTOR = re.compile(r"^\s*(?:(?:public|protected|private)\s+)?([A-Z][\w$]*)\s*\(([^)]*)")
CLASS = re.compile(r"\b(?:class|interface|enum)\s+([A-Za-z_$][\w$]*)")
IDENT = re.compile(r"[A-Za-z_$][\w$]*")
SUB = re.compile(r"[A-Z]+(?![a-z])|[A-Z]?[a-z]+|\d+")
STR = re.compile(r'"((?:[^"\\]|\\.)*)"')

MAX_FILES = 4000
MAX_BYTES = 12_000_000


def subtokens(s):
    return {t.lower() for t in SUB.findall(s) if len(t) >= 3 and t.lower() not in STOP}


def parse(text, base):
    lines = text.splitlines()
    decls = []
    classes = {}
    feats = []
    for i, raw in enumerate(lines, 1):
        s = raw.strip()
        if not s or s.startswith(("*", "/*", "//")):
            feats.append(None)
            continue
        code = raw.split("//", 1)[0]
        ids = {x for x in IDENT.findall(code) if x not in KEYWORDS}
        subs = set()
        for x in ids:
            subs |= subtokens(x)
        for lit in STR.findall(code):
            subs |= subtokens(lit)
        feats.append((frozenset(ids), frozenset(subs)) if (ids or subs) else None)
        m = CLASS.search(code)
        if m:
            classes.setdefault(m.group(1), i)
        m = DECL.match(code)
        if m:
            typ = m.group(1).split("<")[0].split(".")[-1]
            name = m.group(2)
            if typ not in NOT_TYPE and name not in KEYWORDS:
                decls.append((i, name, m.group(3)))
                continue
        m = CTOR.match(code)
        if m and m.group(1) == base:
            decls.append((i, base, m.group(2)))
    return {"lines": lines, "decls": decls, "base": base, "classes": classes, "feats": feats}


def localize(case):
    root = case["root"]
    src = case.get("source_directory", "") or ""
    tst = case.get("test_directory", "") or ""
    report = case.get("report", "") or ""
    failing = case.get("failing_tests", []) or []

    budget = {"files": 0, "bytes": 0}

    def index(base):
        out = {}
        if not base:
            return out
        full = os.path.join(root, base)
        if not os.path.isdir(full):
            return out
        for dp, dn, fn in os.walk(full):
            dn.sort()
            for f in sorted(fn):
                if not f.endswith(".java"):
                    continue
                if budget["files"] >= MAX_FILES or budget["bytes"] >= MAX_BYTES:
                    return out
                p = os.path.join(dp, f)
                try:
                    with open(p, encoding="utf-8", errors="replace") as h:
                        text = h.read(1000000)
                except OSError:
                    continue
                budget["files"] += 1
                budget["bytes"] += len(text)
                rel = os.path.relpath(p, root).replace(os.sep, "/")
                out[rel] = parse(text, f[:-5])
        return out

    prod = index(src)
    tests = index(tst)

    def resolve(fqn, fname, idx):
        top = fqn.split("$")[0]
        suffix = top.replace(".", "/") + ".java"
        matches = [k for k in idx if k == suffix or k.endswith("/" + suffix)]
        if matches:
            return min(matches, key=lambda k: (len(k), k))
        by_name = [k for k in idx if k.endswith("/" + fname)]
        if len(by_name) == 1:
            return by_name[0]
        return None

    def span_bounds(parsed, line):
        lines = parsed["lines"]
        starts = [d[0] for d in parsed["decls"]]
        idx = bisect.bisect_right(starts, line) - 1
        start = starts[idx] if idx >= 0 else max(1, line - 10)
        end = starts[idx + 1] if idx + 1 < len(starts) else len(lines) + 1
        end = min(end, start + 80)
        end = max(end, line + 1)
        return start, end

    def span_text(parsed, line):
        start, end = span_bounds(parsed, line)
        return "\n".join(parsed["lines"][start - 1:end - 1])

    frames = FRAME.findall(report)
    report_ids = set(IDENT.findall(report)) - KEYWORDS
    frame_methods = {m for _, m, _, _ in frames if re.fullmatch(r"[A-Za-z_$][\w$]*", m)}

    A = set()          # identifiers used by the failing tests
    A_strs = []        # string literals in the failing test bodies
    T = set()          # subtokens of failing test method names
    test_classes = set()
    for t in failing:
        cls_part, _, meth = t.partition("::")
        simple = cls_part.split(".")[-1].split("$")[0]
        test_classes.add(simple)
        if not meth:
            continue
        T |= subtokens(meth)
        for k, parsed in tests.items():
            if os.path.basename(k) != simple + ".java":
                continue
            for line, name, _ in parsed["decls"]:
                if name == meth:
                    txt = span_text(parsed, line)
                    A |= {x for x in IDENT.findall(txt) if x not in KEYWORDS}
                    A_strs.extend(STR.findall(txt))
    for fqn, meth, fname, ln in frames:
        rel = resolve(fqn, fname, tests)
        if rel:
            txt = span_text(tests[rel], int(ln))
            A |= {x for x in IDENT.findall(txt) if x not in KEYWORDS}
            A_strs.extend(STR.findall(txt))

    # corpus statistics over production lines
    N = 0
    df_id = Counter()
    df_sub = Counter()
    for parsed in prod.values():
        for f in parsed["feats"]:
            if f:
                N += 1
                df_id.update(f[0])
                df_sub.update(f[1])
    LN = math.log(1 + max(N, 1))

    def wid(x):
        d = df_id.get(x, 0)
        return math.log(1 + N / d) / LN if d else 0.0

    def wsub(x):
        d = df_sub.get(x, 0)
        return math.log(1 + N / d) / LN if d else 0.0

    # query weights
    Q_ids = {}
    for x in A:
        Q_ids[x] = max(Q_ids.get(x, 0.0), 1.0)
    for x in frame_methods:
        Q_ids[x] = max(Q_ids.get(x, 0.0), 1.5)
    for x in report_ids:
        Q_ids[x] = max(Q_ids.get(x, 0.0), 0.3)
    Q_sub = {}
    for x in T:
        Q_sub[x] = max(Q_sub.get(x, 0.0), 1.0)
    for s in A_strs:
        for x in subtokens(s):
            Q_sub[x] = max(Q_sub.get(x, 0.0), 0.6)
    for x in subtokens(report):
        Q_sub[x] = max(Q_sub.get(x, 0.0), 0.3)

    # declaration-level scores
    prod_counts = Counter()
    for parsed in prod.values():
        for _, name, _ in parsed["decls"]:
            prod_counts[name] += 1

    def count(name):
        return max(1, prod_counts.get(name, 1))

    decl_score = {}
    for rel, parsed in prod.items():
        base = parsed["base"]
        for line, name, params in parsed["decls"]:
            c = count(name) ** 0.5
            s = 0.0
            if name in A:
                s += 2.0 / c
            if name in frame_methods:
                s += 1.2 / c
            if name in report_ids:
                s += 0.6 / c
            ov = len(subtokens(name) & T) + 0.5 * len(subtokens(params) & T)
            s += 0.4 * ov
            if base in A:
                s += 0.5
            if base in report_ids:
                s += 0.3
            if s > 0.0:
                decl_score[(rel, line)] = s

    cands = {}

    def add(key, score):
        if score > cands.get(key, -1.0):
            cands[key] = score

    # production stack frames
    rank = 0
    seen = set()
    for fqn, meth, fname, ln in frames:
        rel = resolve(fqn, fname, prod)
        if not rel:
            continue
        key = (rel, int(ln))
        if key in seen:
            continue
        seen.add(key)
        add(key, max(0.3, 2.6 - 0.2 * rank))
        rank += 1

    # line-level and enclosing-declaration scores
    for rel, parsed in prod.items():
        lines = parsed["lines"]
        n = len(lines)
        eb = [0.0] * (n + 2)
        decls = parsed["decls"]
        for k, (l, _, _) in enumerate(decls):
            nxt = decls[k + 1][0] if k + 1 < len(decls) else n + 1
            end = min(nxt, l + 80)
            sc = decl_score.get((rel, l), 0.0) * 0.5
            if sc > 0.0:
                for i in range(l, min(end, n + 1)):
                    if sc > eb[i]:
                        eb[i] = sc
        feats = parsed["feats"]
        for i in range(1, n + 1):
            lp = 0.0
            f = feats[i - 1]
            if f:
                for x in f[0]:
                    m = Q_ids.get(x)
                    if m:
                        lp += m * wid(x)
                for x in f[1]:
                    m = Q_sub.get(x)
                    if m:
                        lp += 0.5 * m * wsub(x)
            total = lp + eb[i] + decl_score.get((rel, i), 0.0)
            if total >= 0.15:
                add((rel, i), total)

        base = parsed["base"]
        if base in A and base in parsed["classes"]:
            add((rel, parsed["classes"][base]), 0.8)
        for tc in test_classes:
            stripped = re.sub(r"Tests?$", "", tc)
            if stripped and base == stripped:
                add((rel, parsed["classes"].get(base, 1)), 0.2)

    # last-resort fallback: file named after the test class
    if not cands:
        for tc in test_classes:
            stripped = re.sub(r"Tests?$", "", tc)
            for rel in prod:
                if os.path.basename(rel) == stripped + ".java":
                    add((rel, 1), 0.1)

    ordered = sorted(cands.items(), key=lambda kv: (-kv[1], kv[0][0], kv[0][1]))
    chosen = []
    for (p, l), _ in ordered:
        if any(p == cp and abs(l - cl) <= 20 for cp, cl in chosen):
            continue
        chosen.append((p, l))
        if len(chosen) == 6:
            break
    return [[p, l] for p, l in chosen]
